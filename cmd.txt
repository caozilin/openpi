# ============================================================
# Linux：安装完整且版本锁定的 OpenPI 环境
# ============================================================
cd /root/openpi

uv sync


# ============================================================
# Linux：将原始数据集转换为 LeRobot 格式
# 输出到 /root/gpufree-data/lerobot_cache/caozilin/franka_mujoco
# 转换与训练共用根项目环境，禁止手工修改 Parquet schema metadata。
# ============================================================
cd /root/openpi

export HF_LEROBOT_HOME=/root/gpufree-data/lerobot_cache

uv run python examples/franka_mujoco/convert_franka_mujoco_data_to_lerobot.py \
  --raw-dir /root/gpufree-data/v1 \
  --repo-id caozilin/franka_mujoco \
  --workers 12 \
  --image-writer-processes 0 \
  --image-writer-threads 2 \
  --progress-interval-seconds 5


# ============================================================
# Linux：为两个配置分别生成归一化参数
# 两个配置读取同一个 caozilin/franka_mujoco 数据集。
# 原始 pi0.5 只统计 7 维动作；TaskTol19 统计 19 维联合目标。
# ============================================================
cd /root/openpi

uv run scripts/compute_franka_mujoco_norm_stats.py \
  --dataset-root /root/gpufree-data/lerobot_cache/caozilin/franka_mujoco


# ============================================================
# Linux：训练原始 pi0.5（只训练 7 维动作）
# ============================================================
cd /root/openpi
mkdir -p /root/gpufree-share/checkpoints

HF_HUB_OFFLINE=1 \
HF_DATASETS_OFFLINE=1 \
CUDA_VISIBLE_DEVICES=0 \
XLA_PYTHON_CLIENT_MEM_FRACTION=0.9 \
uv run --no-sync scripts/train.py pi05_franka_mujoco_joint19 \
  --exp-name=franka_mujoco_joint19_lora \
  --checkpoint-base-dir=/root/gpufree-share/checkpoints \
  --resume \
  --data.lerobot-dataset-root=/root/gpufree-share/datasets/caozilin/franka_mujoco \
  --data.hf-datasets-cache-dir=/root/gpufree-data/openpi_hf_cache \
  --data.lerobot-arrow-cache-dir=/root/gpufree-data/openpi_hf_cache


# ============================================================
# Linux：训练 TaskTol19
# 联合训练 7 维动作、6 维目标旋转和 6 维双侧容差。
# ============================================================
HF_HUB_OFFLINE=1 \
HF_DATASETS_OFFLINE=1 \
CUDA_VISIBLE_DEVICES=0 \
XLA_PYTHON_CLIENT_MEM_FRACTION=0.9 \
uv run --no-sync scripts/train.py pi05_franka_mujoco \
  --exp-name=franka_mujoco_lora \
  --checkpoint-base-dir=/root/gpufree-share/checkpoints \
  --resume \
  --data.lerobot-dataset-root=/root/gpufree-share/datasets/caozilin/franka_mujoco \
  --data.hf-datasets-cache-dir=/root/gpufree-data/openpi_hf_cache \
  --data.lerobot-arrow-cache-dir=/root/gpufree-data/openpi_hf_cache


# ============================================================
# 原有的纯动作 pi05_franka_mujoco 模型服务命令
# ============================================================
cd /root/openpi

CKPT_ROOT=/root/gpufree-share/checkpoints/pi05_franka_mujoco/franka_mujoco_lora
CUDA_VISIBLE_DEVICES=0 \
XLA_PYTHON_CLIENT_MEM_FRACTION=0.9 \
uv run --no-sync scripts/serve_policy.py \
  --port=8000 \
  policy:checkpoint \
  --policy.config=pi05_franka_mujoco \
  --policy.dir="$CKPT_ROOT/10000"

# ============================================================
# TaskTol19 模型服务命令
# ============================================================
cd /root/openpi
CKPT_ROOT=/root/gpufree-share/checkpoints/pi05_franka_mujoco_joint19/franka_mujoco_joint19_lora

CUDA_VISIBLE_DEVICES=0 \
XLA_PYTHON_CLIENT_MEM_FRACTION=0.9 \
uv run --no-sync scripts/serve_policy.py \
  --port=8000 \
  policy:checkpoint \
  --policy.config=pi05_franka_mujoco_joint19 \
  --policy.dir="$CKPT_ROOT/25000"
