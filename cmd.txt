# ============================================================
# Linux：安装完整且版本锁定的 OpenPI 环境
# ============================================================
cd /root/openpi

uv sync


# ============================================================
# Linux：将原始数据集转换为 LeRobot 格式
# 输出到 /root/gpufree-data/lerobot_cache/caozilin/franka_mujoco
# 转换与训练共用根项目环境，禁止手工修改 Parquet schema metadata。
# ============================================================
cd /root/openpi

export HF_LEROBOT_HOME=/root/gpufree-data/lerobot_cache

uv run python examples/franka_mujoco/convert_franka_mujoco_data_to_lerobot.py \
  --raw-dir /root/gpufree-data/v1 \
  --repo-id caozilin/franka_mujoco \
  --workers 12 \
  --image-writer-processes 0 \
  --image-writer-threads 2 \
  --progress-interval-seconds 5


# ============================================================
# Linux：为两个配置分别生成归一化参数
# 两个配置读取同一个 caozilin/franka_mujoco 数据集。
# 原始 pi0.5 只统计 7 维动作；TaskTol19 统计 19 维联合目标。
# ============================================================
uv run scripts/compute_norm_stats.py \
  --config-name pi05_franka_mujoco

uv run scripts/compute_norm_stats.py \
  --config-name pi05_franka_mujoco_joint19


# ============================================================
# Linux：训练原始 pi0.5（只训练 7 维动作）
# ============================================================
CUDA_VISIBLE_DEVICES=0 \
XLA_PYTHON_CLIENT_MEM_FRACTION=0.9 \
uv run scripts/train.py pi05_franka_mujoco \
  --exp-name=franka_mujoco_lora


# ============================================================
# Linux：训练 TaskTol19
# 联合训练 7 维动作、6 维目标旋转和 6 维双侧容差。
# ============================================================
CUDA_VISIBLE_DEVICES=0 \
XLA_PYTHON_CLIENT_MEM_FRACTION=0.9 \
uv run scripts/train.py pi05_franka_mujoco_joint19 \
  --exp-name=franka_mujoco_joint19_lora


# ============================================================
# 原有的纯动作 pi05_franka_mujoco 模型服务命令
# ============================================================
cd /root/openpi

CKPT_ROOT=/root/openpi/checkpoints/pi05_franka_mujoco/franka_mujoco_lora
XLA_PYTHON_CLIENT_MEM_FRACTION=0.9 \
uv run scripts/serve_policy.py \
  --port=8000 \
  policy:checkpoint \
  --policy.config=pi05_franka_mujoco \
  --policy.dir=/root/gpufree-data/checkpoints/pi05_franka_mujoco/franka_mujoco_lora/2000
